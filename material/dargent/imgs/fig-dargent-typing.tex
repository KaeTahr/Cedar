\begin{inductive}{\Q \vdash \ell \sim \tau}
  \inferrule
    {\mathrm{bits}\;(T) = s\\
    T\ \text{is a primitive type} }
    {\Q \vdash \bitrange{o}{s} \sim T}
    {\textsc{PrimTyMatch}}
  \quad
  \inferrule
    {x \sim \tau' \in \Q\\
     \tau' \lesssim_{\ell} \tau}
    {\Q \vdash x\{o\} \sim \tau}
    {\textsc{VarMatch}}
  \\
  \inferrule
    {M\ \text{is the pointer size}\\
     T\ \text{is boxed}}
    {\Q \vdash \bitrange{o}{M} \sim T}
    {\textsc{BoxedTyMatch}}
  \quad
  \inferrule
    { }
    {\Q \vdash () \sim ()}
    {\textsc{UnitMatch}}
  \\
  \inferrule
    {\text{for each $i$: } \Q \vdash \ell_i \sim \tau_i}
    {\Q \vdash \DLRec{\many{\FieldN{f}_i : \ell_i}} \sim \RecordTy{\many{\FieldN{f}_i : \tau_i}}{\Unboxed}}
    {\textsc{URecordMatch}}
  \\
  \inferrule
    {\Q \vdash \ell \sim \tau \\
     \ell\ \text{is byte aligned}}
    {\Q \vdash \DLArr{\ell} \sim \ArrayTy{\tau}{v}{\Unboxed}}
    {\textsc{UArrayMatch}}
  \\
  \inferrule
    {s\ \text{big enough for all alternatives, and smaller than $M$}\\\\
     \text{for each $i$:\ } \Q \vdash \ell_i \sim \tau_i}
    {\Q \vdash \DLVar{s}{\many{\ConsN{C}_i\ (v_i) : \ell_i}} \sim \VariantTy{\many{\ConsN{C}_i\ \tau_i}}}
    {\textsc{VariantMatch}}
    \\
    \inferrule
    { }
    {\Q \vdash \ell \sim \alpha}
    {\textsc{VarTypeMatch}}
    \quad
    \inferrule
    { }
    {\Q \vdash \ell \sim \alpha!}
    {\textsc{BangVarTypeMatch}}
\end{inductive}
